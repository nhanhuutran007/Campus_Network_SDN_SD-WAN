\section{Mục tiêu}

\begin{frame}{Mục tiêu đề tài}
    \begin{block}{Mục tiêu tổng quát}
        Xây dựng mô hình mạng Campus đa cơ sở cho trường đại học trên EVE-NG, kết hợp
        \textbf{SDN} (điều khiển tập trung mạng LAN campus chính) và
        \textbf{SD-WAN} (kết nối thông minh giữa các cơ sở qua Internet và MPLS).
    \end{block}
    \vspace{0.3cm}
    \begin{itemize}
        \item Thiết kế kiến trúc phân cấp Core -- Distribution -- Access, dự phòng VRRP.

              Cần Thơ (200), Đà Nẵng (300), Nha Trang (400) và Site 500 (mở rộng).
        \item Quản lý tập trung 6 switch OVS bằng Ryu Controller qua OpenFlow 1.3, \textbf{tự phục hồi}
              khi mất liên kết, đánh giá theo 5 mục tiêu.
        \item Fabric Cisco SD-WAN (vManage/vSmart/vBond -- Site 900), underlay BGP
              2 nhà cung cấp, \textbf{cảnh báo sự cố qua email}.
        \item Bảo mật và dịch vụ: FW-ASAv HA, DMZ, Server Farm, DNS/DHCP, Syslog, ra Internet có kiểm soát.
    \end{itemize}
\end{frame}

\begin{frame}{SDN -- Kiểm thử khởi động nguội \& cắt link thật (09/10)}
    \footnotesize
    Cắt \textbf{thật} = hạ giao diện ảo trên máy chủ EVE-NG (như rút dây); ping 0,1\,s từ controller tới 4 VLAN.
    \vspace{0.15cm}

    \centering
    \begin{tabular}{@{}p{0.30\textwidth}p{0.30\textwidth}p{0.32\textwidth}@{}}
        \toprule
        \textbf{Tình huống} & \textbf{Trước khi sửa} & \textbf{Sau khi sửa} \\
        \midrule
        Đứt A4-D1 / 4 uplink A*-D1 & mất 1,8 / 2,0\,s & 1,8 / 1,9\,s \\
        Đứt A4-D1 + D2-C1 & VLAN 40 mất 13,8\,s; \alert{kẹt 9/13 link, Dist-SW2 mất controller vĩnh viễn} & tự về 13/13 \\
        Tắt hẳn Dist-SW2 & \alert{4 VLAN mất 100\,\% $\sim$3,5 phút} & \textbf{0--5 gói}; Access nối lại sau $\sim$11\,s \\
        \bottomrule
    \end{tabular}
    \vspace{0.2cm}

    \begin{exampleblock}{4 lỗi đã sửa trong Ryu app (unit test đạt)}
        \footnotesize
        \begin{enumerate}
            \item Access mất controller $\neq$ Access chết: ``đóng băng'' làm lá trên link cha cũ (OVS \texttt{secure} vẫn chạy flow cũ).
            \item Access có thêm đường quản trị dự phòng qua Dist-SW1.
            \item Flow bảo vệ VLAN 99 chỉ \emph{ghi đè}, không xoá trước rồi cài lại.
            \item Ân hạn khi chính controller bị khựng, tránh báo link Core chết giả.
        \end{enumerate}
    \end{exampleblock}
\end{frame}

\begin{frame}{SDN -- Tìm nguyên nhân controller bị khựng}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{Hiện tượng}
                \small
                \begin{itemize}
                    \item Controller khựng theo cụm mỗi 5--15 phút, mỗi lần 0,5--9,1\,s.
                    \item Có cụm làm cả 6 OVS mất kết nối rồi nối lại.
                \end{itemize}
            \end{alertblock}
            \begin{block}{Cách đo}
                \small
                Ghi \texttt{/proc/<pid>/stat} mỗi 0,5\,s trên máy controller (2 vCPU) trong một cụm:
                CPU rảnh về \textbf{0 suốt $\sim$18\,s}; \texttt{java} (ONOS) chiếm $>$ 1 vCPU, Ryu chỉ còn 5--50 tick.
            \end{block}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{exampleblock}{Xử lý}
                \small
                \begin{itemize}
                    \item \textbf{Tắt ONOS} (chỉ dùng để xem topology, Console đã thay thế).
                    \item Ưu tiên CPU cho dịch vụ Ryu (\texttt{CPUShares}).
                    \item RAM controller: 1,3\,GB $\rightarrow$ 179\,MB.
                    \item \texttt{/boot nofail} cho 6 OVS + controller: khởi động đồng loạt không còn vào chế độ khẩn cấp.
                \end{itemize}
            \end{exampleblock}
            \vspace{0.1cm}
            \scriptsize Còn ghi nhận hiếm một lần khựng 4,9\,s sau loạt cắt thật (nghi tài nguyên máy chủ).
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{SDN -- Kênh quản trị VLAN 99 không phụ thuộc STP}
    \begin{columns}[T]
        \begin{column}{0.46\textwidth}
            \begin{alertblock}{Gốc lỗi kịch bản S4/S5}
                \small
                Khung VLAN 99 đi vòng Core-SW1 Et1/2 $\rightarrow$ Dist-SW2 $\rightarrow$ inter-dist
                $\rightarrow$ Dist-SW1 $\rightarrow$ Et0/2: Core nghe \emph{chính BPDU của nó}.
                Khi khôi phục, Et1/2 rơi vào \textbf{RSTP Dispute} $\sim$8\,s $\Rightarrow$ 6 OVS mất controller.
            \end{alertblock}
        \end{column}
        \begin{column}{0.54\textwidth}
            \begin{exampleblock}{Thiết kế mới (chỉ đổi flow do controller cài)}
                \small
                \begin{itemize}
                    \item Khung vào từ cổng Core \textbf{không bao giờ} ra cổng Core khác $\Rightarrow$
                          Et1/2 và Et0/2 đều Desg FWD, STP chỉ là lưới an toàn.
                    \item Mỗi Access tới 4 đường, mỗi Dist 2 đường quản trị \emph{đồng thời}.
                    \item Hai chiều trên inter-dist phân biệt bằng mã VLAN nội bộ.
                    \item Chống phản xạ: drop khung có MAC nguồn là MAC quản trị OVS.
                    \item Không sửa Core-SW1, không sửa script OVS.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
    \vspace{0.15cm}
    \centering\scriptsize Unit test: không vòng, không phản xạ, mọi OVS vẫn tới Core khi mất \emph{bất kỳ} 1 liên kết hoặc 1 Dist. Live: $\sim$100--170\,pps trên Et0/2, Et1/2 (không bão).
\end{frame}
