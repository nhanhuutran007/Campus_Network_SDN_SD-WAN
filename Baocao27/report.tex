\section{Mục tiêu}

\begin{frame}{Mục tiêu đề tài}
    \begin{block}{Mục tiêu tổng quát}
        Xây dựng mô hình mạng Campus đa cơ sở cho trường đại học trên EVE-NG, kết hợp
        \textbf{SDN} (điều khiển tập trung mạng LAN campus chính) và
        \textbf{SD-WAN} (kết nối thông minh giữa các cơ sở qua Internet và MPLS).
    \end{block}
    \vspace{0.3cm}
    \begin{itemize}
        \item Thiết kế kiến trúc phân cấp Core -- Distribution -- Access, dự phòng VRRP.
        \item Quy hoạch VLAN, IP, DHCP cho Site 100 (Campus chính), 3 chi nhánh
              Cần Thơ (200), Đà Nẵng (300), Nha Trang (400) và Site 500 (mới bổ sung).
        \item Quản lý tập trung 6 switch OVS bằng Ryu Controller qua OpenFlow 1.3, đánh giá
              theo 5 mục tiêu: thêm VLAN, khôi phục, hiệu năng, tải thiết bị, chính sách.
        \item Fabric Cisco SD-WAN (vManage/vSmart/vBond -- Site 900), underlay BGP
              2 nhà cung cấp (Internet AS 64511, MPLS AS 64512).
        \item Bảo mật và dịch vụ: FW-ASAv HA, DMZ, Server Farm, DNS/DHCP, Syslog, ra Internet có kiểm soát.
    \end{itemize}
\end{frame}

\section{Khó khăn}

\begin{frame}{Khó khăn gặp phải trong tuần}
    \small
    \begin{itemize}
        \item \textbf{EVE-NG không báo mất link đầu xa:} OVS vẫn thấy cổng \emph{up} khi link bị cắt
              $\Rightarrow$ phải tự thăm dò liên kết bằng probe, chấp nhận gián đoạn 1,8--3,4\,s.
        \item \textbf{Mất kênh điều khiển dễ bị hiểu nhầm là đứt link:} nhiều OVS cùng im lặng
              làm controller chặn cây sai (sự cố D2-C1) $\Rightarrow$ đóng băng link thay vì chặn,
              cắt thử dùng \texttt{hard\_timeout} để OVS tự phục hồi.
        \item \textbf{Vòng lặp default route:} vEdge quảng bá ngược default học từ MPLS lên router
              Internet làm mất Internet toàn lab $\Rightarrow$ lọc \texttt{prefix-list} chiều vào.
        \item \textbf{Hạn chế thiết bị ảo:} ASAv chưa có license (100 kết nối, $\sim$100\,Kbps);
              Core IOL treo khi xoá SVI đang là VRRP master; \texttt{write memory} gộp lệnh không lưu.
        \item \textbf{Tài nguyên máy chủ EVE-NG:} tiến trình vEdge Site 500 bị treo làm máy chủ quá tải
              (load $\sim$96), phải khởi động lại router MPLS bị tắt nhầm.
    \end{itemize}
\end{frame}
